FROM ghcr.io/churchillkaron/avantiqo-code-pod@sha256:21075fc9457c5d6638d9a6a7f04ab748bd4ac50a65fed19e0fd57540a1325262

# V3 is a transport-only overlay on the already-certified immutable V2 runtime.
# handler.py, requirements.txt, vLLM 0.27.1, CUDA/runtime configuration and the
# model contract are inherited byte-for-byte from the base image. CI fails
# closed if handler.py or requirements.txt diverge from the base source SHA.
ENV AVANTIQO_CODE_POD_PORT=8000

WORKDIR /app

COPY services/avantiqo-code-engine/pod_server.py ./pod_server.py
RUN python3 -m py_compile handler.py pod_server.py \
    && python3 -c "from importlib.metadata import version; import runpod, vllm, fastapi, uvicorn; assert version('runpod') == '1.10.0'; assert version('vllm') == '0.27.1'; print('AVANTIQO_CODE_POD_OVERLAY_BUILD_CHECK=PASS')"

# Start the real Pod HTTP server during the image build and prove the canonical
# V3 transport end to end without CUDA inference. The authenticated transport
# probe proves the POST path separately from generation. The queued job then
# uses the engine's benchmark-only runtime probe, so auth, canonical routing,
# queueing, polling, metadata and Uvicorn binding are all covered without
# loading model weights.
RUN AVANTIQO_CODE_POD_TOKEN=avantiqo-code-build-smoke-token-0123456789abcdef python3 - <<'PY'
import json
import os
import subprocess
import time
import urllib.request

token = os.environ["AVANTIQO_CODE_POD_TOKEN"]
base = "http://127.0.0.1:8000"
transport_probe_path = "/v3/transport-probe"
async_submit_path = "/v3/generations"
async_status_template = "/v3/generations/{job_id}"
log_path = "/tmp/avantiqo-code-pod-smoke.log"
log = open(log_path, "w", encoding="utf-8")
process = subprocess.Popen(
    ["python3", "-u", "pod_server.py"],
    env=os.environ.copy(),
    stdout=log,
    stderr=subprocess.STDOUT,
)
try:
    deadline = time.time() + 45
    health = None
    while time.time() < deadline:
        if process.poll() is not None:
            raise RuntimeError(f"pod server exited early: {process.returncode}")
        try:
            with urllib.request.urlopen(f"{base}/health", timeout=2) as response:
                health = json.load(response)
            if health.get("success") is True:
                break
        except Exception:
            time.sleep(1)
    if not health or health.get("success") is not True:
        raise RuntimeError("pod health smoke timeout")
    assert health["contract"] == "AVANTIQO_CODE_POD_HTTP_V3"
    assert health["transport_mode"] == "async-job-polling"
    assert health["transport_probe_path"] == transport_probe_path
    assert health["async_submit_path"] == async_submit_path
    assert health["async_status_path_template"] == async_status_template
    assert health["async_jobs_enabled"] is True
    assert health["synchronous_generation_allowed"] is False
    assert health["engine_loaded"] is False

    probe_request = urllib.request.Request(
        f"{base}{transport_probe_path}",
        data=b"",
        headers={"Authorization": f"Bearer {token}"},
        method="POST",
    )
    with urllib.request.urlopen(probe_request, timeout=5) as response:
        assert response.status == 200
        probe = json.load(response)
    assert probe["success"] is True
    assert probe["contract"] == "AVANTIQO_CODE_POD_HTTP_V3"
    assert probe["transport"] == "pod-http"
    assert probe["transport_mode"] == "async-job-polling"
    assert probe["proxy_timeout_safe"] is True
    assert probe["inference_performed"] is False
    assert probe["raw_reasoning_persisted"] is False

    payload = json.dumps(
        {
            "id": "build-smoke-runtime-probe",
            "input": {
                "contract": "AVANTIQO_CODE_ENGINE_V1",
                "capability": "ai.code.debug",
                "organization_id": "benchmark-only",
                "instruction": "Validate the Pod V3 async transport without loading the model.",
                "structured_specification": {"runtime_probe": True},
            },
        }
    ).encode("utf-8")
    request = urllib.request.Request(
        f"{base}{async_submit_path}",
        data=payload,
        headers={
            "Authorization": f"Bearer {token}",
            "Content-Type": "application/json",
        },
        method="POST",
    )
    with urllib.request.urlopen(request, timeout=5) as response:
        assert response.status == 202
        accepted = json.load(response)
    assert accepted["contract"] == "AVANTIQO_CODE_POD_HTTP_V3"
    assert accepted["proxy_timeout_safe"] is True
    job_id = accepted["job_id"]
    assert accepted["poll_path"] == async_status_template.replace("{job_id}", job_id)

    result = None
    deadline = time.time() + 45
    while time.time() < deadline:
        status_path = async_status_template.replace("{job_id}", job_id)
        request = urllib.request.Request(
            f"{base}{status_path}",
            headers={"Authorization": f"Bearer {token}"},
        )
        with urllib.request.urlopen(request, timeout=5) as response:
            status = json.load(response)
        assert status["contract"] == "AVANTIQO_CODE_POD_HTTP_V3"
        if status["status"] == "FAILED":
            raise RuntimeError(json.dumps(status))
        if status["status"] == "SUCCEEDED":
            result = status
            break
        time.sleep(0.25)
    if result is None:
        raise RuntimeError("pod V3 async job smoke timeout")
    output = result["output"]
    assert output["status"] == "runtime_probe"
    assert output["provider"] == "avantiqo-code"
    assert output["runtime_model"] == "Qwen/Qwen3-Coder-30B-A3B-Instruct-FP8"
    assert output["quantization"] == "fp8"
    assert output["engine_loaded"] is False
    assert output["raw_reasoning_persisted"] is False
    print("AVANTIQO_CODE_POD_V3_TRANSPORT_SMOKE=PASS")
    print("AVANTIQO_CODE_POD_ASYNC_TRANSPORT_SMOKE=PASS")
except Exception:
    log.flush()
    with open(log_path, "r", encoding="utf-8") as handle:
        print(handle.read())
    raise
finally:
    process.terminate()
    try:
        process.wait(timeout=10)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait(timeout=5)
    log.close()
PY

EXPOSE 8000
ENTRYPOINT ["python3", "-u", "pod_server.py"]
