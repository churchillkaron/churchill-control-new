FROM ghcr.io/churchillkaron/avantiqo-code-pod@sha256:21075fc9457c5d6638d9a6a7f04ab748bd4ac50a65fed19e0fd57540a1325262

# V3 is a transport-only overlay on the already-certified immutable V2 runtime.
# The engine, dependencies, CUDA/vLLM runtime and model contract remain inherited
# byte-for-byte. This Dockerfile intentionally executes no runtime during build:
# the external RunPod V3 transport probe is the fail-closed runtime gate before
# any inference is allowed.
ENV AVANTIQO_CODE_POD_PORT=8000

WORKDIR /app

COPY services/avantiqo-code-engine/pod_server.py ./pod_server.py

EXPOSE 8000
ENTRYPOINT ["python3", "-u", "pod_server.py"]
